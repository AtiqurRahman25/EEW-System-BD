
/*
 * ============================================================================
 * Project      : SeismoGuard - Early Earthquake Warning System Bangladesh
 * Platform     : ESP32
 * Sensor       : MPU6050 (Accelerometer and Gyroscope)
 * Display      : 16x2 I2C LCD
 * Communication: Wi-Fi UDP
 *
 * Description:
 * Monitors real-time acceleration data using the MPU6050 sensor, processes
 * motion intensity, detects potential earthquake-like activity, and provides
 * visual, audible, and network-based alerts.
 *
 * ============================================================================
 */

#include <WiFi.h>
#include <WiFiUdp.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <math.h>

// ============================================================================
// 1. NETWORK CONFIGURATION
// ============================================================================

const char* WIFI_SSID = "Mim";
const char* WIFI_PASSWORD = "mim2026*";

const char* UDP_SERVER_ADDRESS = "192.168.0.104";
const uint16_t UDP_SERVER_PORT = 4210;

WiFiUDP udp;

// ============================================================================
// 2. HARDWARE CONFIGURATION
// ============================================================================

// MPU6050 I2C Address
constexpr uint8_t MPU6050_ADDRESS = 0x68;

// LCD Configuration
constexpr uint8_t LCD_ADDRESS = 0x27;
constexpr uint8_t LCD_COLUMNS = 16;
constexpr uint8_t LCD_ROWS = 2;

LiquidCrystal_I2C lcd(
    LCD_ADDRESS,
    LCD_COLUMNS,
    LCD_ROWS
);

// Output Pins
constexpr uint8_t LED_PIN = 2;
constexpr uint8_t BUZZER_PIN = 13;

// ============================================================================
// 3. EARTHQUAKE DETECTION PARAMETERS
// ============================================================================

// Acceleration threshold for motion detection
constexpr float DETECTION_THRESHOLD = 0.8;

// Low-pass filter coefficients
constexpr float FILTER_PREVIOUS_WEIGHT = 0.7;
constexpr float FILTER_CURRENT_WEIGHT = 0.3;

// Detection counter thresholds
constexpr int WARNING_THRESHOLD = 2;
constexpr int EARTHQUAKE_THRESHOLD = 6;

// Sensor update interval
constexpr uint16_t SENSOR_INTERVAL_MS = 100;

// ============================================================================
// 4. GLOBAL VARIABLES
// ============================================================================

float accelerationMagnitude = 0.0;
float previousMagnitude = 0.0;

float motionDifference = 0.0;
float filteredMotionDifference = 0.0;

int earthquakeCounter = 0;

// ============================================================================
// 5. FUNCTION DECLARATIONS
// ============================================================================

void initializeWiFi();
void initializeMPU6050();

bool readMPU6050(
    float &accelerationX,
    float &accelerationY,
    float &accelerationZ
);

void processAcceleration();
String determineEarthquakeStatus();

void updateLED(const String &status);
void updateLCD(const String &status);

void sendUDPData(const String &status);
void triggerEarthquakeAlarm();

// ============================================================================
// 6. SYSTEM INITIALIZATION
// ============================================================================

void setup() {

    Serial.begin(115200);
    delay(1000);

    // Initialize I2C communication
    Wire.begin();

    // Configure output pins
    pinMode(LED_PIN, OUTPUT);
    pinMode(BUZZER_PIN, OUTPUT);

    digitalWrite(LED_PIN, LOW);
    digitalWrite(BUZZER_PIN, LOW);

    // Initialize LCD
    lcd.init();
    lcd.backlight();

    lcd.setCursor(0, 0);
    lcd.print("QuakeSense AI");

    lcd.setCursor(0, 1);
    lcd.print("Initializing...");

    // Initialize network and sensor
    initializeWiFi();
    initializeMPU6050();

    lcd.clear();

    Serial.println();
    Serial.println("================================");
    Serial.println(" QuakeSense AI Initialized");
    Serial.println(" Early Earthquake Warning System");
    Serial.println("================================");

}

// ============================================================================
// 7. WI-FI INITIALIZATION
// ============================================================================

void initializeWiFi() {

    Serial.println("Connecting to Wi-Fi...");

    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

    uint8_t connectionAttempts = 0;

    while (
        WiFi.status() != WL_CONNECTED &&
        connectionAttempts < 20
    ) {

        delay(500);

        Serial.print(".");

        connectionAttempts++;

    }

    lcd.clear();

    if (WiFi.status() == WL_CONNECTED) {

        Serial.println();
        Serial.println("Wi-Fi Connected Successfully");

        Serial.print("IP Address: ");
        Serial.println(WiFi.localIP());

        lcd.print("WiFi Connected");

    } else {

        Serial.println();
        Serial.println("Wi-Fi Connection Failed");

        lcd.print("WiFi Failed");

    }

    delay(1500);

    lcd.clear();

}

// ============================================================================
// 8. MPU6050 INITIALIZATION
// ============================================================================

void initializeMPU6050() {

    Serial.println("Initializing MPU6050...");

    Wire.beginTransmission(MPU6050_ADDRESS);

    // Exit sleep mode
    Wire.write(0x6B);
    Wire.write(0x00);

    byte sensorStatus = Wire.endTransmission(true);

    if (sensorStatus == 0) {

        Serial.println("MPU6050 Initialized Successfully");

    } else {

        Serial.println("MPU6050 Initialization Failed");

    }

}

// ============================================================================
// 9. MPU6050 SENSOR DATA ACQUISITION
// ============================================================================

bool readMPU6050(
    float &accelerationX,
    float &accelerationY,
    float &accelerationZ
) {

    Wire.beginTransmission(MPU6050_ADDRESS);

    // Starting register for accelerometer measurements
    Wire.write(0x3B);

    if (Wire.endTransmission(false) != 0) {
        return false;
    }

    uint8_t bytesReceived = Wire.requestFrom(
        MPU6050_ADDRESS,
        (uint8_t)14,
        (uint8_t)true
    );

    if (bytesReceived != 14) {
        return false;
    }

    // Read raw accelerometer measurements
    int16_t rawAccelerationX =
        (Wire.read() << 8) | Wire.read();

    int16_t rawAccelerationY =
        (Wire.read() << 8) | Wire.read();

    int16_t rawAccelerationZ =
        (Wire.read() << 8) | Wire.read();

    // Skip temperature and gyroscope measurements
    for (uint8_t i = 0; i < 8; i++) {
        Wire.read();
    }

    // Convert raw measurements into acceleration (g)
    accelerationX = rawAccelerationX / 16384.0;
    accelerationY = rawAccelerationY / 16384.0;
    accelerationZ = rawAccelerationZ / 16384.0;

    return true;

}

// ============================================================================
// 10. ACCELERATION PROCESSING AND MOTION ANALYSIS
// ============================================================================

void processAcceleration() {

    float accelerationX = 0.0;
    float accelerationY = 0.0;
    float accelerationZ = 0.0;

    // Acquire sensor measurements
    if (!readMPU6050(
        accelerationX,
        accelerationY,
        accelerationZ
    )) {

        Serial.println("MPU6050 Data Acquisition Failed");
        return;

    }

    // Calculate acceleration magnitude
    accelerationMagnitude = sqrt(
        accelerationX * accelerationX +
        accelerationY * accelerationY +
        accelerationZ * accelerationZ
    );

    // Calculate changes in acceleration magnitude
    motionDifference = fabs(
        accelerationMagnitude - previousMagnitude
    );

    previousMagnitude = accelerationMagnitude;

    // Apply low-pass filtering
    filteredMotionDifference =
        (FILTER_PREVIOUS_WEIGHT * filteredMotionDifference) +
        (FILTER_CURRENT_WEIGHT * motionDifference);

    // Display sensor information
    Serial.print("Motion Difference: ");
    Serial.print(motionDifference);

    Serial.print(" | Filtered Motion: ");
    Serial.println(filteredMotionDifference);

}

// ============================================================================
// 11. EARTHQUAKE DETECTION AND CLASSIFICATION
// ============================================================================

String determineEarthquakeStatus() {

    // Evaluate filtered motion intensity
    if (filteredMotionDifference > DETECTION_THRESHOLD) {

        earthquakeCounter++;

    } else {

        if (earthquakeCounter > 0) {
            earthquakeCounter--;
        }

    }

    // Classify system status
    if (earthquakeCounter > EARTHQUAKE_THRESHOLD) {

        return "EARTHQUAKE!";

    }

    else if (earthquakeCounter > WARNING_THRESHOLD) {

        return "WARNING";

    }

    return "NORMAL";

}

// ============================================================================
// 12. LED STATUS INDICATOR
// ============================================================================

void updateLED(const String &status) {

    if (status == "EARTHQUAKE!") {

        digitalWrite(LED_PIN, HIGH);

    } else {

        digitalWrite(LED_PIN, LOW);

    }

}

// ============================================================================
// 13. LCD STATUS DISPLAY
// ============================================================================

void updateLCD(const String &status) {

    lcd.setCursor(0, 0);
    lcd.print("                ");

    lcd.setCursor(0, 0);
    lcd.print("Status: ");
    lcd.print(status);

    lcd.setCursor(0, 1);
    lcd.print("D:");

    lcd.print(filteredMotionDifference, 3);

    lcd.print("       ");

}

// ============================================================================
// 14. UDP DATA TRANSMISSION
// ============================================================================

void sendUDPData(const String &status) {

    if (WiFi.status() != WL_CONNECTED) {
        return;
    }

    String message =
        status +
        " | D:" +
        String(filteredMotionDifference, 3);

    udp.beginPacket(
        UDP_SERVER_ADDRESS,
        UDP_SERVER_PORT
    );

    udp.print(message);

    udp.endPacket();

    Serial.print("UDP Transmission: ");
    Serial.println(message);

}

// ============================================================================
// 15. EARTHQUAKE ALERT SYSTEM
// ============================================================================

void triggerEarthquakeAlarm() {

    lcd.clear();

    lcd.setCursor(0, 0);
    lcd.print("!!! ALERT !!!");

    lcd.setCursor(0, 1);
    lcd.print("EVACUATE NOW");

    Serial.println("!!! EARTHQUAKE ALERT !!!");

    // Audible warning signal
    for (uint8_t i = 0; i < 5; i++) {

        digitalWrite(BUZZER_PIN, HIGH);
        delay(150);

        digitalWrite(BUZZER_PIN, LOW);
        delay(150);

    }

}

// ============================================================================
// 16. MAIN EXECUTION LOOP
// ============================================================================

void loop() {

    // Acquire and process sensor measurements
    processAcceleration();

    String systemStatus = determineEarthquakeStatus();

    // Update visual indicators
    updateLED(systemStatus);

    if (systemStatus == "EARTHQUAKE!") {

        triggerEarthquakeAlarm();

    } else {

        updateLCD(systemStatus);

    }

    // Transmit real-time monitoring data
    sendUDPData(systemStatus);

    delay(SENSOR_INTERVAL_MS);

}
